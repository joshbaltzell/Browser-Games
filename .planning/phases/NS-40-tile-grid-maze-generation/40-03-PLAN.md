    == renderRoom(room) — full implementation spec ==

    Guard: if (!room || !room.grid) return;  (prevents crash before generateLevel fires)

    Step 1 — Compute centering:
      const originX = Math.floor((W - room.cols * TILE_SIZE) / 2);
      const originY = Math.floor((H - room.rows * TILE_SIZE) / 2);

    Step 2 — Scale-to-fit fallback (only if room exceeds canvas):
      let scale = 1;
      if (room.cols * TILE_SIZE > W || room.rows * TILE_SIZE > H) {
        scale = Math.min(W / (room.cols * TILE_SIZE), H / (room.rows * TILE_SIZE)) * 0.9;
        ctx.save();
        ctx.scale(scale, scale);
        // Recompute origin in scaled space:
        originX = Math.floor((W/scale - room.cols * TILE_SIZE) / 2);
        originY = Math.floor((H/scale - room.rows * TILE_SIZE) / 2);
      }

    Step 3 — Draw floor tiles (tile value 0):
      ctx.fillStyle = "#0d0d1a";
      for each tile (row, col) where grid[row*cols+col] === 0:
        ctx.fillRect(originX + col * TILE_SIZE, originY + row * TILE_SIZE, TILE_SIZE, TILE_SIZE);

    Step 4 — Draw wall tile backgrounds (tile value 1):
      ctx.fillStyle = "#05050c";
      for each tile (row, col) where grid[row*cols+col] === 1:
        ctx.fillRect(originX + col * TILE_SIZE, originY + row * TILE_SIZE, TILE_SIZE, TILE_SIZE);

    Step 5 — Draw wall edge outlines (tile value 1 only):
      ctx.strokeStyle = "rgba(0,229,255,0.35)";
      ctx.lineWidth = 0.5;
      const dirs = [[-1,0],[1,0],[0,-1],[0,1]]; // N, S, W, E offsets
      for each wall tile (row, col) where grid[row*cols+col] === 1:
        ctx.beginPath();
        for each dir [dr, dc] in dirs:
          const nr = row + dr, nc = col + dc;
          const neighborIsFloor = (nr < 0 || nr >= rows || nc < 0 || nc >= cols)
                                   || grid[nr*cols+nc] === 0 || grid[nr*cols+nc] >= 2;
          if (neighborIsFloor):
            const tx = originX + col * TILE_SIZE;
            const ty = originY + row * TILE_SIZE;
            // Draw the edge segment:
            if (dr === -1) { moveTo(tx, ty); lineTo(tx + TILE_SIZE, ty); }              // N edge
            if (dr ===  1) { moveTo(tx, ty + TILE_SIZE); lineTo(tx + TILE_SIZE, ty + TILE_SIZE); } // S edge
            if (dc === -1) { moveTo(tx, ty); lineTo(tx, ty + TILE_SIZE); }              // W edge
            if (dc ===  1) { moveTo(tx + TILE_SIZE, ty); lineTo(tx + TILE_SIZE, ty + TILE_SIZE); } // E edge
        ctx.stroke();

    Step 6 — Draw door glow dots (tile values 2–5):
      const pulse = Math.sin(Date.now() / 800);
      const doorR = 3 + pulse * 1.5;
      const doorColor = (room.type === "BOSS") ? COLORS.pink : COLORS.cyan;
      for each tile (row, col) where grid[row*cols+col] >= 2:
        const cx = originX + col * TILE_SIZE + TILE_SIZE / 2;
        const cy = originY + row * TILE_SIZE + TILE_SIZE / 2;
        glowCircle(cx, cy, doorR, doorColor, 12);

    Step 7 — Conditional debug overlay:
      if (DEBUG && debugOverlayVisible) renderDebugOverlay();

    Step 8 — Restore scale if applied:
      if (scale !== 1) ctx.restore();

    IMPORTANT: The stroke spec in Step 5 draws edge segments using beginPath/moveTo/lineTo
    inside the loop. Batch all edge segments for all wall tiles into a single path before
    calling ctx.stroke() once (performance). Open ctx.beginPath() once before the outer
    wall loop, add all moveTo/lineTo calls inside, then stroke once after.

    Decision coverage: D-16, D-17, D-18, D-19 (partial — renderDebugOverlay in Task 2).

    function renderDebugOverlay() {
      // Guard: only callable when currentLevel is populated
      if (!currentLevel) return;
      const rooms = currentLevel.rooms;

      // Slot grid dimensions
      const SLOT_W = W / 4;
      const SLOT_H = H / 4;

      // Layer 1: 4x4 slot grid lines
      ctx.save();
      ctx.strokeStyle = "rgba(255,255,255,0.08)";
      ctx.lineWidth = 1;
      for (let r = 0; r <= 4; r++) {
        ctx.beginPath();
        ctx.moveTo(0, r * SLOT_H);
        ctx.lineTo(W, r * SLOT_H);
        ctx.stroke();
      }
      for (let c = 0; c <= 4; c++) {
        ctx.beginPath();
        ctx.moveTo(c * SLOT_W, 0);
        ctx.lineTo(c * SLOT_W, H);
        ctx.stroke();
      }

      // Layer 2: Filled slot rects by room type
      // Colors: START=#00e5ff20, NORMAL=rgba(255,255,255,0.04), ELITE=#b388ff20, SHOP=#fffb9620, BOSS=#ff3b6b20
      const typeColors = {
        "START": "rgba(0,229,255,0.125)",
        "NORMAL": "rgba(255,255,255,0.04)",
        "ELITE": "rgba(179,136,255,0.125)",
        "SHOP": "rgba(255,251,150,0.125)",
        "BOSS": "rgba(255,59,107,0.125)",
      };
      rooms.forEach(function(room) {
        const sr = Math.floor(room.slotIdx / 4);
        const sc = room.slotIdx % 4;
        ctx.fillStyle = typeColors[room.type] || "rgba(255,255,255,0.04)";
        ctx.fillRect(sc * SLOT_W, sr * SLOT_H, SLOT_W, SLOT_H);
      });

      // Layer 3: Connection lines between slot centers
      ctx.globalAlpha = 0.4;
      ctx.strokeStyle = COLORS.cyan;
      ctx.lineWidth = 1;
      currentLevel.connections.forEach(function(connStr) {
        const parts = connStr.split("-");
        const idxA = parseInt(parts[0], 10);
        const idxB = parseInt(parts[1], 10);
        const rA = Math.floor(idxA / 4), cA = idxA % 4;
        const rB = Math.floor(idxB / 4), cB = idxB % 4;
        ctx.beginPath();
        ctx.moveTo(cA * SLOT_W + SLOT_W / 2, rA * SLOT_H + SLOT_H / 2);
        ctx.lineTo(cB * SLOT_W + SLOT_W / 2, rB * SLOT_H + SLOT_H / 2);
        ctx.stroke();
      });
      ctx.globalAlpha = 1;

      // Layer 4: Room-type labels at slot centers
      ctx.font = "11px \"Segoe UI\", system-ui, sans-serif";
      ctx.textAlign = "center";
      ctx.textBaseline = "middle";
      ctx.fillStyle = "rgba(255,255,255,0.6)";
      rooms.forEach(function(room) {
        const sr = Math.floor(room.slotIdx / 4);
        const sc = room.slotIdx % 4;
        ctx.fillText(room.type, sc * SLOT_W + SLOT_W / 2, sr * SLOT_H + SLOT_H / 2);
      });

      // Layer 5: Current room highlight (white stroke rect around current room's slot)
      if (currentRoom) {
        const sr = Math.floor(currentRoom.slotIdx / 4);
        const sc = currentRoom.slotIdx % 4;
        ctx.strokeStyle = "rgba(255,255,255,0.4)";
        ctx.lineWidth = 1.5;
        ctx.strokeRect(sc * SLOT_W + 2, sr * SLOT_H + 2, SLOT_W - 4, SLOT_H - 4);
      }

      // Layer 6: Seed readout at top-left
      ctx.font = "12px \"Segoe UI\", system-ui, sans-serif";
      ctx.textAlign = "left";
      ctx.textBaseline = "top";
      ctx.fillStyle = "rgba(154,154,192,0.8)";
      ctx.fillText("SEED: " + currentLevel.seed, 8, 8);

      ctx.restore();
    }
